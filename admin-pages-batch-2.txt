'use client'

import React, { useState, useEffect } from 'react'
import { Project } from '@/types/database'
import { supabase } from '@/lib/supabaseClient'
import { FeedbackAlert } from '@/components/admin/FeedbackAlert'
import { DeleteConfirmModal } from '@/components/admin/DeleteConfirmModal'
import { RecordModal } from '@/components/admin/RecordModal'
import {
  FolderGit2,
  Plus,
  Edit2,
  Trash2,
  ExternalLink,
  Code2,
  RotateCcw,
  Eye,
  EyeOff,
  Star
} from 'lucide-react'

const initialProjectForm: Partial<Project> = {
  title: '',
  slug: '',
  short_description: '',
  full_description: '',
  category: 'Data Analytics',
  status: 'completed',
  is_featured: false,
  image_url: '',
  demo_url: '',
  github_url: '',
  technologies: [],
  metrics_highlights: [],
  display_order: 0,
  published: true
}

export default function AdminProjectsPage() {
  const [projects, setProjects] = useState<Project[]>([])
  const [loading, setLoading] = useState(true)
  const [feedback, setFeedback] = useState<{ type: 'success' | 'error' | 'loading'; message: string } | null>(null)

  const [modalOpen, setModalOpen] = useState(false)
  const [formData, setFormData] = useState<Partial<Project>>(initialProjectForm)
  const [technologiesText, setTechnologiesText] = useState('')
  const [metricsText, setMetricsText] = useState('')
  const [saving, setSaving] = useState(false)

  const [deletingProject, setDeletingProject] = useState<Project | null>(null)
  const [deleting, setDeleting] = useState(false)

  const fetchProjects = async () => {
    setLoading(true)
    try {
      const { data, error } = await supabase
        .from('projects')
        .select('*')
        .order('display_order', { ascending: true })
        .order('created_at', { ascending: false })

      if (error) {
        console.error('Error fetching projects:', error)
        setFeedback({ type: 'error', message: `Failed to load projects: ${error.message}` })
      } else {
        setProjects(data || [])
      }
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Unknown load error'
      setFeedback({ type: 'error', message: msg })
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    fetchProjects()
  }, [])

  const handleOpenAdd = () => {
    setFormData({
      ...initialProjectForm,
      display_order: projects.length + 1
    })
    setTechnologiesText('')
    setMetricsText('')
    setModalOpen(true)
  }

  const handleOpenEdit = (project: Project) => {
    setFormData(project)
    setTechnologiesText((project.technologies || []).join(', '))
    setMetricsText((project.metrics_highlights || []).join('\n'))
    setModalOpen(true)
  }

  const handleSlugify = (title: string) => {
    return title
      .toLowerCase()
      .trim()
      .replace(/[^a-z0-9\s-]/g, '')
      .replace(/\s+/g, '-')
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!formData.title || !formData.short_description || !formData.category) {
      setFeedback({ type: 'error', message: 'Title, Short Description, and Category are required.' })
      return
    }

    setSaving(true)
    try {
      const autoSlug = formData.slug?.trim() || handleSlugify(formData.title)
      const parsedTech = technologiesText
        .split(',')
        .map((s) => s.trim())
        .filter(Boolean)
      const parsedMetrics = metricsText
        .split('\n')
        .map((s) => s.trim())
        .filter(Boolean)

      const payload = {
        title: formData.title,
        slug: autoSlug,
        short_description: formData.short_description,
        full_description: formData.full_description || null,
        category: formData.category,
        status: formData.status || 'completed',
        is_featured: formData.is_featured || false,
        image_url: formData.image_url || null,
        demo_url: formData.demo_url || null,
        github_url: formData.github_url || null,
        technologies: parsedTech,
        metrics_highlights: parsedMetrics,
        display_order: Number(formData.display_order) || 0,
        published: formData.published !== false,
        updated_at: new Date().toISOString()
      }

      if (formData.id) {
        const { data, error } = await supabase
          .from('projects')
          .update(payload)
          .eq('id', formData.id)
          .select()
          .single()

        if (error) throw error

        setProjects((prev) => prev.map((p) => (p.id === formData.id ? data : p)))
        setFeedback({ type: 'success', message: `Project "${data.title}" updated successfully!` })
      } else {
        const { data, error } = await supabase
          .from('projects')
          .insert(payload)
          .select()
          .single()

        if (error) throw error

        setProjects((prev) => [...prev, data])
        setFeedback({ type: 'success', message: `Project "${data.title}" created successfully!` })
      }

      setModalOpen(false)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Database error'
      setFeedback({ type: 'error', message: `Could not save project: ${msg}` })
    } finally {
      setSaving(false)
    }
  }

  const handleDelete = async () => {
    if (!deletingProject) return
    setDeleting(true)

    try {
      const { error } = await supabase
        .from('projects')
        .delete()
        .eq('id', deletingProject.id)

      if (error) throw error

      setProjects((prev) => prev.filter((p) => p.id !== deletingProject.id))
      setFeedback({ type: 'success', message: `Project "${deletingProject.title}" deleted successfully.` })
      setDeletingProject(null)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Delete failed'
      setFeedback({ type: 'error', message: `Could not delete project: ${msg}` })
    } finally {
      setDeleting(false)
    }
  }

  const handleTogglePublish = async (project: Project) => {
    const updatedStatus = !project.published
    try {
      const { error } = await supabase
        .from('projects')
        .update({ published: updatedStatus, updated_at: new Date().toISOString() })
        .eq('id', project.id)

      if (error) throw error

      setProjects((prev) =>
        prev.map((p) => (p.id === project.id ? { ...p, published: updatedStatus } : p))
      )
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Status toggle failed'
      setFeedback({ type: 'error', message: `Could not change published status: ${msg}` })
    }
  }

  return (
    <div className="space-y-6 max-w-6xl pb-12">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 pb-4 border-b border-slate-800">
        <div>
          <h1 className="text-2xl font-bold text-white flex items-center gap-2">
            <FolderGit2 className="w-6 h-6 text-blue-500" />
            Projects Manager
          </h1>
          <p className="text-xs text-slate-400 mt-1">
            Create, edit, organize display orders, and publish portfolio showcase projects.
          </p>
        </div>

        <div className="flex items-center gap-2">
          <button
            type="button"
            onClick={fetchProjects}
            className="px-3 py-2 rounded-xl bg-slate-900 border border-slate-800 hover:bg-slate-800 text-slate-300 text-xs font-semibold flex items-center gap-1.5 transition-colors"
          >
            <RotateCcw className="w-3.5 h-3.5" /> Reload
          </button>
          <button
            type="button"
            onClick={handleOpenAdd}
            className="px-4 py-2 rounded-xl bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold flex items-center gap-1.5 shadow-lg shadow-blue-600/20 transition-all"
          >
            <Plus className="w-4 h-4" /> Add New Project
          </button>
        </div>
      </div>

      {feedback && (
        <FeedbackAlert
          type={feedback.type}
          message={feedback.message}
          onClose={() => setFeedback(null)}
        />
      )}

      <div className="space-y-4">
        {loading ? (
          <FeedbackAlert type="loading" message="Loading projects from Supabase database..." />
        ) : projects.length === 0 ? (
          <div className="p-8 rounded-2xl bg-slate-900 border border-slate-800 text-center space-y-3">
            <FolderGit2 className="w-10 h-10 text-slate-600 mx-auto" />
            <p className="text-sm font-semibold text-white">No projects found in database</p>
            <p className="text-xs text-slate-400">Click &ldquo;Add New Project&rdquo; to add your first work sample.</p>
          </div>
        ) : (
          <div className="grid grid-cols-1 gap-4">
            {projects.map((project) => (
              <div
                key={project.id}
                className={`p-5 rounded-2xl bg-slate-900 border transition-all ${
                  project.published
                    ? 'border-slate-800 hover:border-slate-700'
                    : 'border-slate-800/50 opacity-70 bg-slate-950/40'
                } flex flex-col md:flex-row md:items-center justify-between gap-4`}
              >
                <div className="space-y-2 flex-1">
                  <div className="flex flex-wrap items-center gap-2">
                    <span className="text-[11px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                      Order: {project.display_order}
                    </span>
                    <span className="text-[11px] font-semibold px-2 py-0.5 rounded-full bg-blue-950 text-blue-300 border border-blue-800">
                      {project.category}
                    </span>
                    <span className={`text-[11px] font-semibold px-2 py-0.5 rounded-full capitalize ${
                      project.status === 'completed'
                        ? 'bg-emerald-950 text-emerald-300 border border-emerald-800'
                        : 'bg-amber-950 text-amber-300 border border-amber-800'
                    }`}>
                      {project.status.replace('_', ' ')}
                    </span>
                    {project.is_featured && (
                      <span className="text-[11px] font-semibold px-2 py-0.5 rounded-full bg-amber-950 text-amber-300 border border-amber-800 flex items-center gap-1">
                        <Star className="w-3 h-3 fill-amber-300" /> Featured
                      </span>
                    )}
                  </div>

                  <div>
                    <h3 className="text-base font-bold text-white flex items-center gap-2">
                      {project.title}
                    </h3>
                    <p className="text-xs text-slate-400 line-clamp-2 mt-1">
                      {project.short_description}
                    </p>
                  </div>

                  {project.technologies && project.technologies.length > 0 && (
                    <div className="flex flex-wrap items-center gap-1.5 pt-1">
                      {project.technologies.map((t, idx) => (
                        <span key={idx} className="text-[10px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                          {t}
                        </span>
                      ))}
                    </div>
                  )}
                </div>

                <div className="flex items-center gap-2 shrink-0 pt-2 md:pt-0 border-t md:border-t-0 border-slate-800">
                  <button
                    type="button"
                    onClick={() => handleTogglePublish(project)}
                    className={`px-3 py-1.5 rounded-xl text-xs font-semibold flex items-center gap-1.5 transition-colors border ${
                      project.published
                        ? 'bg-emerald-950/60 text-emerald-300 border-emerald-800 hover:bg-emerald-900/60'
                        : 'bg-slate-950 text-slate-500 border-slate-800 hover:text-slate-300'
                    }`}
                  >
                    {project.published ? (
                      <>
                        <Eye className="w-3.5 h-3.5" /> Published
                      </>
                    ) : (
                      <>
                        <EyeOff className="w-3.5 h-3.5" /> Draft
                      </>
                    )}
                  </button>

                  {project.demo_url && (
                    <a
                      href={project.demo_url}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="p-2 rounded-xl bg-slate-950 hover:bg-slate-800 text-slate-400 hover:text-white border border-slate-800 transition-colors"
                      title="Demo URL"
                    >
                      <ExternalLink className="w-3.5 h-3.5" />
                    </a>
                  )}

                  {project.github_url && (
                    <a
                      href={project.github_url}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="p-2 rounded-xl bg-slate-950 hover:bg-slate-800 text-slate-400 hover:text-white border border-slate-800 transition-colors"
                      title="GitHub Repository"
                    >
                      <Code2 className="w-3.5 h-3.5" />
                    </a>
                  )}

                  <button
                    type="button"
                    onClick={() => handleOpenEdit(project)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-blue-950 text-blue-400 hover:text-blue-300 border border-slate-800 hover:border-blue-800 transition-colors"
                    title="Edit project"
                  >
                    <Edit2 className="w-3.5 h-3.5" />
                  </button>

                  <button
                    type="button"
                    onClick={() => setDeletingProject(project)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-rose-950 text-rose-400 hover:text-rose-300 border border-slate-800 hover:border-rose-800 transition-colors"
                    title="Delete project"
                  >
                    <Trash2 className="w-3.5 h-3.5" />
                  </button>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>

      <RecordModal
        isOpen={modalOpen}
        title={formData.id ? 'Edit Project' : 'Add New Project'}
        subtitle="Manage project details, slug, technologies, URLs, and showcase status"
        maxWidth="max-w-3xl"
        onClose={() => setModalOpen(false)}
      >
        <form onSubmit={handleSubmit} className="space-y-4">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Project Title <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"
                required
                value={formData.title || ''}
                onChange={(e) => {
                  const title = e.target.value
                  setFormData((prev) => ({
                    ...prev,
                    title,
                    slug: prev.id ? prev.slug : handleSlugify(title)
                  }))
                }}
                placeholder="e.g. Student Analytics Dashboard"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                URL Slug <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"
                required
                value={formData.slug || ''}
                onChange={(e) => setFormData({ ...formData, slug: e.target.value })}
                placeholder="student-analytics-dashboard"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm font-mono focus:border-blue-500 focus:outline-none"
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Category <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"
                required
                value={formData.category || ''}
                onChange={(e) => setFormData({ ...formData, category: e.target.value })}
                placeholder="Data Analytics, AI Solution, etc."
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Status
              </label>
              <select
                value={formData.status || 'completed'}
                onChange={(e) => setFormData({ ...formData, status: e.target.value as 'completed' | 'in_progress' })}
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              >
                <option value="completed">Completed</option>
                <option value="in_progress">In Progress</option>
              </select>
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Display Order
              </label>
              <input
                type="number"
                value={formData.display_order ?? 0}

            </div>
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-300 mb-1.5">
              Short Description <span className="text-rose-400">*</span>
            </label>
            <textarea
              rows={2}
              required
              value={formData.short_description || ''}
              onChange={(e) => setFormData({ ...formData, short_description: e.target.value })}
              placeholder="High-level overview of the project shown on project cards..."
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-300 mb-1.5">
              Full Description / Methodology
            </label>
            <textarea
              rows={4}
              value={formData.full_description || ''}
              onChange={(e) => setFormData({ ...formData, full_description: e.target.value })}
              placeholder="Detailed architecture, findings, business impact..."
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
            />
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Technologies (comma-separated)
              </label>
              <input
                type="text"
                value={technologiesText}
                onChange={(e) => setTechnologiesText(e.target.value)}
                placeholder="Python, Power BI, SQL, Pandas"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Image / Cover URL
              </label>
              <input
                type="text"
                value={formData.image_url || ''}
                onChange={(e) => setFormData({ ...formData, image_url: e.target.value })}
                placeholder="https://images.unsplash.com/... or uploaded image URL"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Live Demo / Dashboard URL
              </label>
              <input
                type="url"
                value={formData.demo_url || ''}
                onChange={(e) => setFormData({ ...formData, demo_url: e.target.value })}
                placeholder="https://..."
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                GitHub Repository URL
              </label>
              <input
                type="url"
                value={formData.github_url || ''}
                onChange={(e) => setFormData({ ...formData, github_url: e.target.value })}
                placeholder="https://github.com/..."
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>
          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-300 mb-1.5">
              Key Metrics / Impact Highlights (one per line)
            </label>
            <textarea
              rows={3}
              value={metricsText}
              onChange={(e) => setMetricsText(e.target.value)}
              placeholder="e.g. 98% accuracy in customer retention\nDecreased query latency by 45%"
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
            />
          </div>

          <div className="flex flex-wrap items-center gap-6 pt-2">
            <label className="flex items-center gap-2 cursor-pointer">
              <input
                type="checkbox"
                checked={formData.is_featured || false}
                onChange={(e) => setFormData({ ...formData, is_featured: e.target.checked })}
                className="w-4 h-4 rounded text-blue-600 bg-slate-950 border-slate-800 focus:ring-0"
              />
              <span className="text-xs font-semibold text-slate-300">Feature this project prominently</span>
            </label>

            <label className="flex items-center gap-2 cursor-pointer">
              <input
                type="checkbox"
                checked={formData.published !== false}
                onChange={(e) => setFormData({ ...formData, published: e.target.checked })}
                className="w-4 h-4 rounded text-blue-600 bg-slate-950 border-slate-800 focus:ring-0"
              />
              <span className="text-xs font-semibold text-slate-300">Publish on public portfolio</span>
            </label>
          </div>

          <div className="flex items-center justify-end gap-3 pt-4 border-t border-slate-800">
            <button
              type="button"
              disabled={saving}
              onClick={() => setModalOpen(false)}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-slate-400 hover:text-white"
            >
              Cancel
            </button>
            <button
              type="submit"
              disabled={saving}
              className="px-5 py-2.5 rounded-xl bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold shadow-lg shadow-blue-600/20"
            >
              {saving ? 'Saving...' : formData.id ? 'Save Changes' : 'Create Project'}
            </button>
          </div>
        </form>
      </RecordModal>

      <DeleteConfirmModal
        isOpen={!!deletingProject}
        title="Delete Project"
        description="Are you sure you want to permanently delete this project from the database?"
        itemName={deletingProject?.title}
        loading={deleting}
        onConfirm={handleDelete}
        onCancel={() => setDeletingProject(null)}
      />
    </div>
  )
}

  display_order: 0,

export default function AdminEducationPage() {
  const [activeTab, setActiveTab] = useState<'education' | 'certifications'>('education')
  const [educations, setEducations] = useState<Education[]>([])
  const [certifications, setCertifications] = useState<Certification[]>([])
  const [loading, setLoading] = useState(true)
  const [feedback, setFeedback] = useState<{ type: 'success' | 'error' | 'loading'; message: string } | null>(null)

  const [eduModalOpen, setEduModalOpen] = useState(false)
  const [eduForm, setEduForm] = useState<Partial<Education>>(initialEducationForm)
  const [savingEdu, setSavingEdu] = useState(false)
  const [deletingEdu, setDeletingEdu] = useState<Education | null>(null)

  const [certModalOpen, setCertModalOpen] = useState(false)
  const [certForm, setCertForm] = useState<Partial<Certification>>(initialCertForm)
  const [savingCert, setSavingCert] = useState(false)
  const [deletingCert, setDeletingCert] = useState<Certification | null>(null)

  const [deleting, setDeleting] = useState(false)

  const fetchData = async () => {
    setLoading(true)
    try {
      const [eduRes, certRes] = await Promise.all([
        supabase.from('education').select('*').order('display_order', { ascending: true }).order('start_year', { ascending: false }),
        supabase.from('certifications').select('*').order('display_order', { ascending: true }).order('issue_date', { ascending: false })
      ])

      if (eduRes.error) throw eduRes.error
      if (certRes.error) throw certRes.error

      setEducations(eduRes.data || [])
      setCertifications(certRes.data || [])
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Unknown load error'
      setFeedback({ type: 'error', message: `Failed to load records: ${msg}` })
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    fetchData()
  }, [])

  const handleOpenAddEdu = () => {
    setEduForm({
      ...initialEducationForm,
      display_order: educations.length + 1
    })
    setEduModalOpen(true)
  }

  const handleOpenEditEdu = (edu: Education) => {
    setEduForm(edu)
    setEduModalOpen(true)
  }

  const handleSaveEdu = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!eduForm.degree || !eduForm.field_of_study || !eduForm.institution || !eduForm.start_year) {
      setFeedback({ type: 'error', message: 'Degree, Field of Study, Institution, and Start Year are required.' })
      return
    }

    setSavingEdu(true)
    try {
      const payload = {
        degree: eduForm.degree,
        field_of_study: eduForm.field_of_study,
        institution: eduForm.institution,
        location: eduForm.location || null,
        start_year: Number(eduForm.start_year),
        end_year: eduForm.is_current ? null : (eduForm.end_year ? Number(eduForm.end_year) : null),
        is_current: eduForm.is_current || false,
        grade_or_honors: eduForm.grade_or_honors || null,
        activities: eduForm.activities || null,
        display_order: Number(eduForm.display_order) || 0,
        published: eduForm.published !== false
      }

      if (eduForm.id) {
        const { data, error } = await supabase
          .from('education')
          .update(payload)
          .eq('id', eduForm.id)
          .select()
          .single()

        if (error) throw error
        setEducations((prev) => prev.map((e) => (e.id === eduForm.id ? data : e)))
        setFeedback({ type: 'success', message: `Education record updated successfully!` })
      } else {
        const { data, error } = await supabase
          .from('education')
          .insert(payload)
          .select()
          .single()

        if (error) throw error
        setEducations((prev) => [...prev, data])
        setFeedback({ type: 'success', message: `Education record created successfully!` })
      }
      setEduModalOpen(false)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Could not save education: ${msg}` })
    } finally {
      setSavingEdu(false)
    }
  }

  const handleDeleteEdu = async () => {
    if (!deletingEdu) return
    setDeleting(true)
    try {
      const { error } = await supabase.from('education').delete().eq('id', deletingEdu.id)
      if (error) throw error

      setEducations((prev) => prev.filter((e) => e.id !== deletingEdu.id))
      setFeedback({ type: 'success', message: `Education record deleted.` })
      setDeletingEdu(null)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Delete failed: ${msg}` })
    } finally {
      setDeleting(false)
    }
  }

  const handleOpenAddCert = () => {
    setCertForm({
      ...initialCertForm,
      display_order: certifications.length + 1
    })
    setCertModalOpen(true)
  }

  const handleOpenEditCert = (cert: Certification) => {
    setCertForm(cert)
    setCertModalOpen(true)
  }

  const handleSaveCert = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!certForm.title || !certForm.issuing_organization || !certForm.issue_date) {
      setFeedback({ type: 'error', message: 'Title, Issuing Organization, and Issue Date are required.' })
      return
    }

    setSavingCert(true)
    try {
      const payload = {
        title: certForm.title,
        issuing_organization: certForm.issuing_organization,
        issue_date: certForm.issue_date,
        expiration_date: certForm.expiration_date || null,
        credential_id: certForm.credential_id || null,
        credential_url: certForm.credential_url || null,
        display_order: Number(certForm.display_order) || 0,
        published: certForm.published !== false
      }

      if (certForm.id) {
        const { data, error } = await supabase
          .from('certifications')
          .update(payload)
          .eq('id', certForm.id)
          .select()
          .single()

        if (error) throw error
        setCertifications((prev) => prev.map((c) => (c.id === certForm.id ? data : c)))
        setFeedback({ type: 'success', message: `Certification "${data.title}" updated!` })
      } else {
        const { data, error } = await supabase
          .from('certifications')
          .insert(payload)
          .select()
          .single()

        if (error) throw error
        setCertifications((prev) => [...prev, data])
        setFeedback({ type: 'success', message: `Certification "${data.title}" added!` })
      }
      setCertModalOpen(false)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Could not save certification: ${msg}` })
    } finally {
      setSavingCert(false)
    }
  }

  const handleDeleteCert = async () => {
    if (!deletingCert) return
    setDeleting(true)
    try {
      const { error } = await supabase.from('certifications').delete().eq('id', deletingCert.id)
      if (error) throw error

      setCertifications((prev) => prev.filter((c) => c.id !== deletingCert.id))
      setFeedback({ type: 'success', message: `Certification "${deletingCert.title}" deleted.` })
      setDeletingCert(null)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Delete failed: ${msg}` })
    } finally {
      setDeleting(false)
    }
  }

  return (
    <div className="space-y-6 max-w-6xl pb-12">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 pb-4 border-b border-slate-800">
        <div>
          <h1 className="text-2xl font-bold text-white flex items-center gap-2">
            <GraduationCap className="w-6 h-6 text-blue-500" />
            Education &amp; Certifications Manager
          </h1>
          <p className="text-xs text-slate-400 mt-1">
            Manage academic degrees, institutions, honors, and verified technical credentials.
          </p>
        </div>

        <div className="flex items-center gap-2">
          <button
            type="button"
            onClick={fetchData}
            className="px-3 py-2 rounded-xl bg-slate-900 border border-slate-800 hover:bg-slate-800 text-slate-300 text-xs font-semibold flex items-center gap-1.5 transition-colors"
          >
            <RotateCcw className="w-3.5 h-3.5" /> Reload
          </button>
          <button
            type="button"
            onClick={activeTab === 'education' ? handleOpenAddEdu : handleOpenAddCert}
            className="px-4 py-2 rounded-xl bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold flex items-center gap-1.5 shadow-lg shadow-blue-600/20 transition-all"
          >
            <Plus className="w-4 h-4" /> Add {activeTab === 'education' ? 'Education' : 'Certification'}
          </button>
        </div>
      </div>

      {feedback && (
        <FeedbackAlert
          type={feedback.type}
          message={feedback.message}
          onClose={() => setFeedback(null)}
        />
      )}

      <div className="flex items-center gap-3 border-b border-slate-800 pb-2">
        <button
          type="button"
          onClick={() => setActiveTab('education')}
          className={`px-4 py-2 rounded-xl text-xs font-semibold flex items-center gap-2 transition-all ${
            activeTab === 'education'
              ? 'bg-blue-600 text-white shadow-md shadow-blue-600/20'
              : 'text-slate-400 hover:text-white hover:bg-slate-900'
          }`}
        >
          <GraduationCap className="w-4 h-4" /> Academic Education ({educations.length})
        </button>
        <button
          type="button"
          onClick={() => setActiveTab('certifications')}
          className={`px-4 py-2 rounded-xl text-xs font-semibold flex items-center gap-2 transition-all ${
            activeTab === 'certifications'
              ? 'bg-blue-600 text-white shadow-md shadow-blue-600/20'
              : 'text-slate-400 hover:text-white hover:bg-slate-900'
          }`}
        >
          <Award className="w-4 h-4" /> Certifications &amp; Credentials ({certifications.length})
        </button>
      </div>

      {activeTab === 'education' && (
        <div className="space-y-4">
          {loading ? (
            <FeedbackAlert type="loading" message="Loading education records..." />
          ) : educations.length === 0 ? (
            <div className="p-8 rounded-2xl bg-slate-900 border border-slate-800 text-center space-y-3">
              <GraduationCap className="w-10 h-10 text-slate-600 mx-auto" />
              <p className="text-sm font-semibold text-white">No academic degrees found</p>
              <p className="text-xs text-slate-400">Click &ldquo;Add Education&rdquo; to insert your alma mater.</p>
            </div>
          ) : (
            <div className="grid grid-cols-1 gap-4">
              {educations.map((edu) => (
                <div
                  key={edu.id}
                  className={`p-5 rounded-2xl bg-slate-900 border transition-all ${
                    edu.published
                      ? 'border-slate-800 hover:border-slate-700'
                      : 'border-slate-800/50 opacity-70 bg-slate-950/40'
                  } flex flex-col md:flex-row md:items-center justify-between gap-4`}
                >
                  <div className="space-y-1.5 flex-1">
                    <div className="flex items-center gap-2">
                      <span className="text-[11px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                        Order: {edu.display_order}
                      </span>
                      {edu.grade_or_honors && (
                        <span className="text-[11px] font-semibold px-2 py-0.5 rounded-full bg-blue-950 text-blue-300 border border-blue-800">
                          {edu.grade_or_honors}
                        </span>
                      )}
                    </div>

                    <h3 className="text-base font-bold text-white">{edu.degree}</h3>
                    <p className="text-xs font-semibold text-blue-400">{edu.field_of_study}</p>
                    <p className="text-xs text-slate-400 flex items-center gap-1.5">
                      <Building2 className="w-3.5 h-3.5 text-slate-500" />
                      {edu.institution} {edu.location && `â€¢ ${edu.location}`}
                    </p>
                    <p className="text-[11px] text-slate-500 flex items-center gap-1.5">
                      <Calendar className="w-3 h-3" />
                      {edu.start_year} &ndash; {edu.is_current ? 'Present' : edu.end_year || 'Completed'}
                    </p>
                  </div>

                  <div className="flex items-center gap-2 shrink-0">
                    <button
                      type="button"
                      onClick={async () => {
                        const newStatus = !edu.published
                        await supabase.from('education').update({ published: newStatus }).eq('id', edu.id)
                        setEducations((prev) => prev.map((e) => (e.id === edu.id ? { ...e, published: newStatus } : e)))
                      }}
                      className={`px-3 py-1.5 rounded-xl text-xs font-semibold flex items-center gap-1.5 transition-colors border ${
                        edu.published
                          ? 'bg-emerald-950/60 text-emerald-300 border-emerald-800'
                          : 'bg-slate-950 text-slate-500 border-slate-800'
                      }`}
                    >
                      {edu.published ? <Eye className="w-3.5 h-3.5" /> : <EyeOff className="w-3.5 h-3.5" />}
                      {edu.published ? 'Published' : 'Draft'}
                    </button>

                    <button
                      type="button"
                      onClick={() => handleOpenEditEdu(edu)}
                      className="p-2 rounded-xl bg-slate-950 hover:bg-blue-950 text-blue-400 hover:text-blue-300 border border-slate-800 transition-colors"
                      title="Edit"
                    >
                      <Edit2 className="w-3.5 h-3.5" />
                    </button>

                    <button
                      type="button"
                      onClick={() => setDeletingEdu(edu)}
                      className="p-2 rounded-xl bg-slate-950 hover:bg-rose-950 text-rose-400 hover:text-rose-300 border border-slate-800 transition-colors"
                      title="Delete"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {activeTab === 'certifications' && (
        <div className="space-y-4">
          {loading ? (
            <FeedbackAlert type="loading" message="Loading certifications..." />
          ) : certifications.length === 0 ? (
            <div className="p-8 rounded-2xl bg-slate-900 border border-slate-800 text-center space-y-3">
              <Award className="w-10 h-10 text-slate-600 mx-auto" />
              <p className="text-sm font-semibold text-white">No certifications recorded</p>
              <p className="text-xs text-slate-400">Click &ldquo;Add Certification&rdquo; to insert verified credentials.</p>
            </div>
          ) : (
            <div className="grid grid-cols-1 gap-4">
              {certifications.map((cert) => (
                <div
                  key={cert.id}
                  className={`p-5 rounded-2xl bg-slate-900 border transition-all ${
                    cert.published
                      ? 'border-slate-800 hover:border-slate-700'
                      : 'border-slate-800/50 opacity-70 bg-slate-950/40'
                  } flex flex-col md:flex-row md:items-center justify-between gap-4`}
                >
                  <div className="space-y-1.5 flex-1">
                    <div className="flex items-center gap-2">
                      <span className="text-[11px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                        Order: {cert.display_order}
                      </span>
                      {cert.credential_id && (
                        <span className="text-[11px] font-mono text-slate-400">
                          ID: {cert.credential_id}
                        </span>
                      )}
                    </div>

                    <h3 className="text-base font-bold text-white">{cert.title}</h3>
                    <p className="text-xs font-semibold text-blue-400">{cert.issuing_organization}</p>
                    <p className="text-[11px] text-slate-400 flex items-center gap-1.5">
                      <Calendar className="w-3 h-3 text-slate-500" />
                      Issued: {cert.issue_date} {cert.expiration_date && `â€¢ Expires: ${cert.expiration_date}`}
                    </p>
                  </div>

                  <div className="flex items-center gap-2 shrink-0">
                    <button
                      type="button"
                      onClick={async () => {
                        const newStatus = !cert.published
                        await supabase.from('certifications').update({ published: newStatus }).eq('id', cert.id)
                        setCertifications((prev) => prev.map((c) => (c.id === cert.id ? { ...c, published: newStatus } : c)))
                      }}
                      className={`px-3 py-1.5 rounded-xl text-xs font-semibold flex items-center gap-1.5 transition-colors border ${
                        cert.published
                          ? 'bg-emerald-950/60 text-emerald-300 border-emerald-800'
                          : 'bg-slate-950 text-slate-500 border-slate-800'
                      }`}
                    >
                      {cert.published ? <Eye className="w-3.5 h-3.5" /> : <EyeOff className="w-3.5 h-3.5" />}
                      {cert.published ? 'Published' : 'Draft'}
                    </button>

                    {cert.credential_url && (
                      <a
                        href={cert.credential_url}
                        target="_blank"
                        rel="noopener noreferrer"
                        className="p-2 rounded-xl bg-slate-950 hover:bg-slate-800 text-slate-400 hover:text-white border border-slate-800 transition-colors"
                        title="Verification Link"
                      >
                        <ExternalLink className="w-3.5 h-3.5" />
                      </a>
                    )}

                    <button
                      type="button"
                      onClick={() => handleOpenEditCert(cert)}
                      className="p-2 rounded-xl bg-slate-950 hover:bg-blue-950 text-blue-400 hover:text-blue-300 border border-slate-800 transition-colors"
                      title="Edit"
                    >
                      <Edit2 className="w-3.5 h-3.5" />
                    </button>

                    <button
                      type="button"
                      onClick={() => setDeletingCert(cert)}
                      className="p-2 rounded-xl bg-slate-950 hover:bg-rose-950 text-rose-400 hover:text-rose-300 border border-slate-800 transition-colors"
                      title="Delete"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      <RecordModal
        isOpen={eduModalOpen}
        title={eduForm.id ? 'Edit Education Record' : 'Add Education Record'}
        subtitle="Manage academic degree, study field, university, and completion years"
        maxWidth="max-w-2xl"
        onClose={() => setEduModalOpen(false)}
      >
        <form onSubmit={handleSaveEdu} className="space-y-4">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Degree / Qualification <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"
                required
                value={eduForm.degree || ''}
                onChange={(e) => setEduForm({ ...eduForm, degree: e.target.value })}
                placeholder="e.g. Bachelor of Technology (B.Tech)"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Field of Study / Major <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"
                required
                value={eduForm.field_of_study || ''}
                onChange={(e) => setEduForm({ ...eduForm, field_of_study: e.target.value })}
                placeholder="e.g. Mathematics / Computer Science"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Institution <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"
                required
                value={eduForm.institution || ''}
                onChange={(e) => setEduForm({ ...eduForm, institution: e.target.value })}
                placeholder="e.g. LAUTECH"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Location
              </label>
              <input
                type="text"
                value={eduForm.location || ''}
                onChange={(e) => setEduForm({ ...eduForm, location: e.target.value })}
                placeholder="e.g. Oyo State, Nigeria"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Start Year <span className="text-rose-400">*</span>
              </label>
              <input
                type="number"
                required
                value={eduForm.start_year || ''}
                onChange={(e) => setEduForm({ ...eduForm, start_year: Number(e.target.value) })}
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                End Year {!eduForm.is_current && <span className="text-slate-400">(or expected)</span>}
              </label>
              <input
                type="number"
                disabled={eduForm.is_current}
                value={eduForm.end_year || ''}
                onChange={(e) => setEduForm({ ...eduForm, end_year: Number(e.target.value) })}
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none disabled:opacity-40"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Display Order
              </label>
              <input
                type="number"
                value={eduForm.display_order ?? 0}
                onChange={(e) => setEduForm({ ...eduForm, display_order: Number(e.target.value) })}
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Grade / Honors
              </label>
              <input
                type="text"
                value={eduForm.grade_or_honors || ''}
                onChange={(e) => setEduForm({ ...eduForm, grade_or_honors: e.target.value })}
                placeholder="First Class Honors, Distinction, etc."
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Activities / Societies
              </label>
              <input
                type="text"
                value={eduForm.activities || ''}
                onChange={(e) => setEduForm({ ...eduForm, activities: e.target.value })}
                placeholder="Lead Math Tutor, Society President..."
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>
          </div>

          <div className="flex items-center gap-6">
            <label className="flex items-center gap-2 cursor-pointer">
              <input
                type="checkbox"
                checked={eduForm.is_current || false}
                onChange={(e) => setEduForm({ ...eduForm, is_current: e.target.checked })}
                className="w-4 h-4 rounded text-blue-600 bg-slate-950 border-slate-800 focus:ring-0"
              />
              <span className="text-xs font-semibold text-slate-300">Currently studying here</span>
            </label>

            <label className="flex items-center gap-2 cursor-pointer">
              <input
                type="checkbox"
                checked={eduForm.published !== false}
                onChange={(e) => setEduForm({ ...eduForm, published: e.target.checked })}
                className="w-4 h-4 rounded text-blue-600 bg-slate-950 border-slate-800 focus:ring-0"
              />
              <span className="text-xs font-semibold text-slate-300">Publish on public portfolio</span>

import React, { useState, useEffect } from 'react'
import { TeachingMentorship } from '@/types/database'
import { supabase } from '@/lib/supabaseClient'
import { FeedbackAlert } from '@/components/admin/FeedbackAlert'
import { DeleteConfirmModal } from '@/components/admin/DeleteConfirmModal'
import { RecordModal } from '@/components/admin/RecordModal'
import {
  GraduationCap,
  Plus,
  Edit2,
  Trash2,
  RotateCcw,
  Eye,
  EyeOff,
  Users,
  Calendar
} from 'lucide-react'

const initialTeachingForm: Partial<TeachingMentorship> = {
  title: '',
  organization: '',
  role: '',
  period: '',
  summary: '',
  impact_highlights: [],
  students_impacted: 0,
  display_order: 0,
  published: true
}

export default function AdminTeachingPage() {
  const [items, setItems] = useState<TeachingMentorship[]>([])
  const [loading, setLoading] = useState(true)
  const [feedback, setFeedback] = useState<{ type: 'success' | 'error' | 'loading'; message: string } | null>(null)

  const [modalOpen, setModalOpen] = useState(false)
  const [formData, setFormData] = useState<Partial<TeachingMentorship>>(initialTeachingForm)
  const [highlightsText, setHighlightsText] = useState('')
  const [saving, setSaving] = useState(false)

  const [deletingItem, setDeletingItem] = useState<TeachingMentorship | null>(null)
  const [deleting, setDeleting] = useState(false)

  const fetchData = async () => {
    setLoading(true)
    try {
      const { data, error } = await supabase
        .from('teaching_mentorship')
        .select('*')
        .order('display_order', { ascending: true })
        .order('created_at', { ascending: false })

      if (error) throw error
      setItems(data || [])
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Unknown load error'
      setFeedback({ type: 'error', message: `Load failed: ${msg}` })
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    fetchData()
  }, [])

  const handleOpenAdd = () => {
    setFormData({
      ...initialTeachingForm,
      display_order: items.length + 1
    })
    setHighlightsText('')
    setModalOpen(true)
  }

  const handleOpenEdit = (item: TeachingMentorship) => {
    setFormData(item)
    setHighlightsText((item.impact_highlights || []).join('\n'))
    setModalOpen(true)
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!formData.title || !formData.organization || !formData.role || !formData.period || !formData.summary) {
      setFeedback({ type: 'error', message: 'Title, Organization, Role, Period, and Summary are required.' })
      return
    }

    setSaving(true)
    try {
      const parsedHighlights = highlightsText
        .split('\n')
        .map((s) => s.trim())
        .filter(Boolean)

      const payload = {
        title: formData.title,
        organization: formData.organization,
        role: formData.role,
        period: formData.period,
        summary: formData.summary,
        impact_highlights: parsedHighlights,
        students_impacted: Number(formData.students_impacted) || 0,
        display_order: Number(formData.display_order) || 0,
        published: formData.published !== false
      }

      if (formData.id) {
        const { data, error } = await supabase
          .from('teaching_mentorship')
          .update(payload)
          .eq('id', formData.id)
          .select()
          .single()

        if (error) throw error
        setItems((prev) => prev.map((i) => (i.id === formData.id ? data : i)))
        setFeedback({ type: 'success', message: `Teaching engagement updated!` })
      } else {
        const { data, error } = await supabase
          .from('teaching_mentorship')
          .insert(payload)
          .select()
          .single()

        if (error) throw error
        setItems((prev) => [...prev, data])
        setFeedback({ type: 'success', message: `Teaching engagement added!` })
      }
      setModalOpen(false)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Could not save: ${msg}` })
    } finally {
      setSaving(false)
    }
  }

  const handleDelete = async () => {
    if (!deletingItem) return
    setDeleting(true)
    try {
      const { error } = await supabase.from('teaching_mentorship').delete().eq('id', deletingItem.id)
      if (error) throw error

      setItems((prev) => prev.filter((i) => i.id !== deletingItem.id))
      setFeedback({ type: 'success', message: `Record deleted successfully.` })
      setDeletingItem(null)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Delete failed: ${msg}` })
    } finally {
      setDeleting(false)
    }
  }

  return (
    <div className="space-y-6 max-w-6xl pb-12">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 pb-4 border-b border-slate-800">
        <div>
          <h1 className="text-2xl font-bold text-white flex items-center gap-2">
            <GraduationCap className="w-6 h-6 text-blue-500" />
            Teaching &amp; Mentorship Manager
          </h1>
          <p className="text-xs text-slate-400 mt-1">
            Manage instructional courses, pedagogy engagements, student impact counts, and highlights.
          </p>
        </div>

        <div className="flex items-center gap-2">
          <button
            type="button"
            onClick={fetchData}
            className="px-3 py-2 rounded-xl bg-slate-900 border border-slate-800 hover:bg-slate-800 text-slate-300 text-xs font-semibold flex items-center gap-1.5 transition-colors"
          >
            <RotateCcw className="w-3.5 h-3.5" /> Reload
          </button>
          <button
            type="button"
            onClick={handleOpenAdd}
            className="px-4 py-2 rounded-xl bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold flex items-center gap-1.5 shadow-lg shadow-blue-600/20 transition-all"
          >
            <Plus className="w-4 h-4" /> Add Teaching Engagement
          </button>
        </div>
      </div>

      {feedback && (
        <FeedbackAlert
          type={feedback.type}
          message={feedback.message}
          onClose={() => setFeedback(null)}
        />
      )}

      <div className="space-y-4">
        {loading ? (
          <FeedbackAlert type="loading" message="Loading teaching engagements..." />
        ) : items.length === 0 ? (
          <div className="p-8 rounded-2xl bg-slate-900 border border-slate-800 text-center space-y-3">
            <GraduationCap className="w-10 h-10 text-slate-600 mx-auto" />
            <p className="text-sm font-semibold text-white">No teaching engagements found</p>
            <p className="text-xs text-slate-400">Click &ldquo;Add Teaching Engagement&rdquo; to insert your courses.</p>
          </div>
        ) : (
          <div className="grid grid-cols-1 gap-4">
            {items.map((item) => (
              <div
                key={item.id}
                className={`p-5 rounded-2xl bg-slate-900 border transition-all ${
                  item.published
                    ? 'border-slate-800 hover:border-slate-700'
                    : 'border-slate-800/50 opacity-70 bg-slate-950/40'
                } flex flex-col md:flex-row md:items-start justify-between gap-4`}
              >
                <div className="space-y-2 flex-1">
                  <div className="flex flex-wrap items-center gap-2">
                    <span className="text-[11px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                      Order: {item.display_order}
                    </span>
                    {item.students_impacted > 0 && (
                      <span className="text-[11px] font-semibold px-2 py-0.5 rounded-full bg-blue-950 text-blue-300 border border-blue-800 flex items-center gap-1">

                      </span>
                    )}
                  </div>

                  <div>

                    <p className="text-xs font-semibold text-blue-400">{item.role} â€¢ {item.organization}</p>

                      setItems((prev) => prev.map((i) => (i.id === item.id ? { ...i, published: newStatus } : i)))
                    }}
                    className={`px-3 py-1.5 rounded-xl text-xs font-semibold flex items-center gap-1.5 transition-colors border ${

                        ? 'bg-emerald-950/60 text-emerald-300 border-emerald-800'
                        : 'bg-slate-950 text-slate-500 border-slate-800'
                    }`}

                    {item.published ? <Eye className="w-3.5 h-3.5" /> : <EyeOff className="w-3.5 h-3.5" />}
                    {item.published ? 'Published' : 'Draft'}
                  </button>

                  <button
                    type="button"
                    onClick={() => handleOpenEdit(item)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-blue-950 text-blue-400 hover:text-blue-300 border border-slate-800 transition-colors"
                    title="Edit"
                  >
                    <Edit2 className="w-3.5 h-3.5" />
                  </button>

                  <button
                    type="button"
                    onClick={() => setDeletingItem(item)}

          </div>
        )}

      <RecordModal

import { supabase } from '@/lib/supabaseClient'
import { FeedbackAlert } from '@/components/admin/FeedbackAlert'
import { DeleteConfirmModal } from '@/components/admin/DeleteConfirmModal'
import { RecordModal } from '@/components/admin/RecordModal'
import {
  Users2,
  Plus,
  Edit2,
  Trash2,
  RotateCcw,
  Eye,
  EyeOff,
  Calendar
} from 'lucide-react'

const initialLeadershipForm: Partial<LeadershipCommunity> = {
  title: '',
  organization: '',
  role: '',
  period: '',
  summary: '',
  achievements: [],
  display_order: 0,
  published: true
}

export default function AdminLeadershipPage() {
  const [items, setItems] = useState<LeadershipCommunity[]>([])
  const [loading, setLoading] = useState(true)
  const [feedback, setFeedback] = useState<{ type: 'success' | 'error' | 'loading'; message: string } | null>(null)

  const [modalOpen, setModalOpen] = useState(false)
  const [formData, setFormData] = useState<Partial<LeadershipCommunity>>(initialLeadershipForm)
  const [achievementsText, setAchievementsText] = useState('')
  const [saving, setSaving] = useState(false)

  const [deletingItem, setDeletingItem] = useState<LeadershipCommunity | null>(null)
  const [deleting, setDeleting] = useState(false)

  const fetchData = async () => {
    setLoading(true)
    try {
      const { data, error } = await supabase
        .from('leadership_community')
        .select('*')
        .order('display_order', { ascending: true })
        .order('created_at', { ascending: false })

      if (error) throw error
      setItems(data || [])
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Unknown load error'
      setFeedback({ type: 'error', message: `Load failed: ${msg}` })
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    fetchData()
  }, [])

  const handleOpenAdd = () => {
    setFormData({
      ...initialLeadershipForm,
      display_order: items.length + 1
    })
    setAchievementsText('')
    setModalOpen(true)
  }

  const handleOpenEdit = (item: LeadershipCommunity) => {
    setFormData(item)
    setAchievementsText((item.achievements || []).join('\n'))
    setModalOpen(true)
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!formData.title || !formData.organization || !formData.role || !formData.period || !formData.summary) {
      setFeedback({ type: 'error', message: 'Title, Organization, Role, Period, and Summary are required.' })
      return
    }

    setSaving(true)
    try {
      const parsedAchievements = achievementsText
        .split('\n')
        .map((s) => s.trim())
        .filter(Boolean)

      const payload = {
        title: formData.title,
        organization: formData.organization,
        role: formData.role,
        period: formData.period,
        summary: formData.summary,
        achievements: parsedAchievements,
        display_order: Number(formData.display_order) || 0,
        published: formData.published !== false
      }

      if (formData.id) {
        const { data, error } = await supabase
          .from('leadership_community')
          .update(payload)
          .eq('id', formData.id)
          .select()
          .single()

        if (error) throw error
        setItems((prev) => prev.map((i) => (i.id === formData.id ? data : i)))
        setFeedback({ type: 'success', message: `Leadership initiative updated!` })
      } else {
        const { data, error } = await supabase
          .from('leadership_community')
          .insert(payload)
          .select()
          .single()

        if (error) throw error
        setItems((prev) => [...prev, data])
        setFeedback({ type: 'success', message: `Leadership initiative added!` })
      }
      setModalOpen(false)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Could not save: ${msg}` })
    } finally {
      setSaving(false)
    }
  }

  const handleDelete = async () => {
    if (!deletingItem) return
    setDeleting(true)
    try {
      const { error } = await supabase.from('leadership_community').delete().eq('id', deletingItem.id)
      if (error) throw error

      setItems((prev) => prev.filter((i) => i.id !== deletingItem.id))
      setFeedback({ type: 'success', message: `Record deleted successfully.` })
      setDeletingItem(null)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Delete failed: ${msg}` })
    } finally {
      setDeleting(false)
    }
  }

  return (
    <div className="space-y-6 max-w-6xl pb-12">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 pb-4 border-b border-slate-800">
        <div>
          <h1 className="text-2xl font-bold text-white flex items-center gap-2">
            <Users2 className="w-6 h-6 text-blue-500" />
            Leadership &amp; Community Forum Manager
          </h1>
          <p className="text-xs text-slate-400 mt-1">
            Manage your community initiatives, Creative Thinkers Forum outreach, and leadership appointments.
          </p>
        </div>

        <div className="flex items-center gap-2">
          <button
            type="button"
            onClick={fetchData}
            className="px-3 py-2 rounded-xl bg-slate-900 border border-slate-800 hover:bg-slate-800 text-slate-300 text-xs font-semibold flex items-center gap-1.5 transition-colors"
          >
            <RotateCcw className="w-3.5 h-3.5" /> Reload
          </button>
          <button
            type="button"
            onClick={handleOpenAdd}
            className="px-4 py-2 rounded-xl bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold flex items-center gap-1.5 shadow-lg shadow-blue-600/20 transition-all"
          >
            <Plus className="w-4 h-4" /> Add Leadership Initiative
          </button>
        </div>
      </div>

      {feedback && (
        <FeedbackAlert
          type={feedback.type}
          message={feedback.message}
          onClose={() => setFeedback(null)}
        />
      )}

      <div className="space-y-4">
        {loading ? (
          <FeedbackAlert type="loading" message="Loading leadership initiatives..." />
        ) : items.length === 0 ? (
          <div className="p-8 rounded-2xl bg-slate-900 border border-slate-800 text-center space-y-3">
            <Users2 className="w-10 h-10 text-slate-600 mx-auto" />
            <p className="text-sm font-semibold text-white">No leadership initiatives found</p>
            <p className="text-xs text-slate-400">Click &ldquo;Add Leadership Initiative&rdquo; to insert your work with Creative Thinkers Forum.</p>
          </div>
        ) : (
          <div className="grid grid-cols-1 gap-4">
            {items.map((item) => (
              <div
                key={item.id}
                className={`p-5 rounded-2xl bg-slate-900 border transition-all ${
                  item.published
                    ? 'border-slate-800 hover:border-slate-700'
                    : 'border-slate-800/50 opacity-70 bg-slate-950/40'
                } flex flex-col md:flex-row md:items-start justify-between gap-4`}
              >
                <div className="space-y-2 flex-1">
                  <div className="flex items-center gap-2">
                    <span className="text-[11px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                      Order: {item.display_order}
                    </span>
                  </div>

                  <div>
                    <h3 className="text-base font-bold text-white">{item.title}</h3>
                    <p className="text-xs font-semibold text-blue-400">{item.role} â€¢ {item.organization}</p>
                    <p className="text-[11px] text-slate-400 flex items-center gap-1.5 mt-0.5">
                      <Calendar className="w-3.5 h-3.5 text-slate-500" /> {item.period}
                    </p>
                  </div>

                  <p className="text-xs text-slate-300 leading-relaxed">{item.summary}</p>

                  {item.achievements && item.achievements.length > 0 && (
                    <ul className="text-xs text-slate-400 list-disc list-inside space-y-0.5 pt-1">
                      {item.achievements.map((a, idx) => (
                        <li key={idx}>{a}</li>
                      ))}
                    </ul>
                  )}
                </div>

                <div className="flex items-center gap-2 shrink-0 pt-2 md:pt-0 border-t md:border-t-0 border-slate-800">
                  <button
                    type="button"
                    onClick={async () => {
                      const newStatus = !item.published
                      await supabase.from('leadership_community').update({ published: newStatus }).eq('id', item.id)
                      setItems((prev) => prev.map((i) => (i.id === item.id ? { ...i, published: newStatus } : i)))
                    }}
                    className={`px-3 py-1.5 rounded-xl text-xs font-semibold flex items-center gap-1.5 transition-colors border ${
                      item.published
                        ? 'bg-emerald-950/60 text-emerald-300 border-emerald-800'
                        : 'bg-slate-950 text-slate-500 border-slate-800'
                    }`}
                  >
                    {item.published ? <Eye className="w-3.5 h-3.5" /> : <EyeOff className="w-3.5 h-3.5" />}
                    {item.published ? 'Published' : 'Draft'}
                  </button>

                  <button
                    type="button"
                    onClick={() => handleOpenEdit(item)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-blue-950 text-blue-400 hover:text-blue-300 border border-slate-800 transition-colors"
                    title="Edit"
                  >
                    <Edit2 className="w-3.5 h-3.5" />
                  </button>

                  <button
                    type="button"
                    onClick={() => setDeletingItem(item)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-rose-950 text-rose-400 hover:text-rose-300 border border-slate-800 transition-colors"
                    title="Delete"
                  >
                    <Trash2 className="w-3.5 h-3.5" />
                  </button>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>

      <RecordModal
        isOpen={modalOpen}
        title={formData.id ? 'Edit Leadership Initiative' : 'Add Leadership Initiative'}
        subtitle="Manage program title, community organization, role, and key milestones"
        maxWidth="max-w-2xl"

                value={formData.organization || ''}

                placeholder="e.g. Creative Thinkers Forum"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"

                required
                value={formData.role || ''}
                onChange={(e) => setFormData({ ...formData, role: e.target.value })}
                placeholder="e.g. Founder & Convener"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Period <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"
                required
                value={formData.period || ''}
                onChange={(e) => setFormData({ ...formData, period: e.target.value })}
                placeholder="e.g. 2022 - Present"
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Display Order
              </label>

                type="number"
                value={formData.display_order ?? 0}
                onChange={(e) => setFormData({ ...formData, display_order: Number(e.target.value) })}
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"

              required
              value={formData.summary || ''}

              placeholder="Initiative background, community target, goals..."

          </div>

          <div>
            <label className="block text-xs font-semibold text-slate-300 mb-1.5">
              Key Achievements & Milestones (one per line)

              rows={3}
              value={achievementsText}
              onChange={(e) => setAchievementsText(e.target.value)}
              placeholder="e.g. Organized free technical workshops on data literacy\nBuilt active community network of young innovators"
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
            />
          </div>

          <div>
            <label className="flex items-center gap-2 cursor-pointer">
              <input
                type="checkbox"
                checked={formData.published !== false}
                onChange={(e) => setFormData({ ...formData, published: e.target.checked })}
                className="w-4 h-4 rounded text-blue-600 bg-slate-950 border-slate-800 focus:ring-0"
              />
              <span className="text-xs font-semibold text-slate-300">Publish on public portfolio</span>
            </label>
          </div>

          <div className="flex items-center justify-end gap-3 pt-4 border-t border-slate-800">
            <button
              type="button"
              disabled={saving}
              onClick={() => setModalOpen(false)}
              className="px-4 py-2 rounded-xl text-xs font-semibold text-slate-400 hover:text-white"
            >
              Cancel
            </button>
            <button
              type="submit"
              disabled={saving}
              className="px-5 py-2.5 rounded-xl bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold shadow-lg shadow-blue-600/20"

              {saving ? 'Saving...' : formData.id ? 'Save Changes' : 'Create Record'}

          </div>

        isOpen={!!deletingItem}
        title="Delete Leadership Record"
        description="Are you sure you want to delete this leadership entry from the database?"
        itemName={deletingItem?.title}
        loading={deleting}
        onConfirm={handleDelete}
        onCancel={() => setDeletingItem(null)}
      />
    </div>
  )
}

                    <p className="text-xs text-slate-400 flex items-center gap-1.5">
                      <Users className="w-3.5 h-3.5 text-slate-500" /> {item.authors.join(', ')}
                    </p>
                  )}

                  <p className="text-xs text-slate-300 leading-relaxed line-clamp-3">{item.abstract}</p>
                </div>

                <div className="flex items-center gap-2 shrink-0 pt-2 md:pt-0 border-t md:border-t-0 border-slate-800">
                  <button
                    type="button"
                    onClick={async () => {
                      const newStatus = !item.published
                      await supabase.from('publications').update({ published: newStatus }).eq('id', item.id)
                      setItems((prev) => prev.map((i) => (i.id === item.id ? { ...i, published: newStatus } : i)))
                    }}
                    className={`px-3 py-1.5 rounded-xl text-xs font-semibold flex items-center gap-1.5 transition-colors border ${
                      item.published
                        ? 'bg-emerald-950/60 text-emerald-300 border-emerald-800'

                    }`}
                  >
                    {item.published ? <Eye className="w-3.5 h-3.5" /> : <EyeOff className="w-3.5 h-3.5" />}
                    {item.published ? 'Published' : 'Draft'}
                  </button>

                  {item.doi_or_url && (
                    <a
                      href={item.doi_or_url}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="p-2 rounded-xl bg-slate-950 hover:bg-slate-800 text-slate-400 hover:text-white border border-slate-800 transition-colors"
                      title="Open Paper / DOI"
                    >
                      <ExternalLink className="w-3.5 h-3.5" />
                    </a>
                  )}

                  <button
                    type="button"
                    onClick={() => handleOpenEdit(item)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-blue-950 text-blue-400 hover:text-blue-300 border border-slate-800 transition-colors"
                    title="Edit"
                  >
                    <Edit2 className="w-3.5 h-3.5" />
                  </button>

                  <button
                    type="button"
                    onClick={() => setDeletingItem(item)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-rose-950 text-rose-400 hover:text-rose-300 border border-slate-800 transition-colors"

      <RecordModal

              placeholder="e.g. Optimizing Predictive Analytics in Technical Education"
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"

'use client'

import React, { useState, useEffect } from 'react'
import { Achievement } from '@/types/database'
import { supabase } from '@/lib/supabaseClient'
import { FeedbackAlert } from '@/components/admin/FeedbackAlert'
import { DeleteConfirmModal } from '@/components/admin/DeleteConfirmModal'
import { RecordModal } from '@/components/admin/RecordModal'
import {
  Trophy,
  Plus,
  Edit2,
  Trash2,
  RotateCcw,
  Eye,
  EyeOff,
  Calendar
} from 'lucide-react'

const initialAchievementForm: Partial<Achievement> = {
  title: '',
  category: 'Teaching Excellence',
  issuer: '',
  award_date: '',
  description: '',
  display_order: 0,
  published: true
}

export default function AdminAchievementsPage() {
  const [items, setItems] = useState<Achievement[]>([])
  const [loading, setLoading] = useState(true)
  const [feedback, setFeedback] = useState<{ type: 'success' | 'error' | 'loading'; message: string } | null>(null)

  const [modalOpen, setModalOpen] = useState(false)
  const [formData, setFormData] = useState<Partial<Achievement>>(initialAchievementForm)
  const [saving, setSaving] = useState(false)

  const [deletingItem, setDeletingItem] = useState<Achievement | null>(null)
  const [deleting, setDeleting] = useState(false)

  const fetchData = async () => {
    setLoading(true)
    try {
      const { data, error } = await supabase
        .from('achievements')
        .select('*')
        .order('display_order', { ascending: true })
        .order('award_date', { ascending: false })

      if (error) throw error
      setItems(data || [])
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Unknown load error'
      setFeedback({ type: 'error', message: `Load failed: ${msg}` })
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    fetchData()
  }, [])

  const handleOpenAdd = () => {
    setFormData({
      ...initialAchievementForm,
      display_order: items.length + 1
    })
    setModalOpen(true)
  }

  const handleOpenEdit = (item: Achievement) => {
    setFormData(item)
    setModalOpen(true)
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!formData.title || !formData.category || !formData.issuer || !formData.award_date || !formData.description) {
      setFeedback({ type: 'error', message: 'Title, Category, Issuer, Award Date, and Description are required.' })
      return
    }

    setSaving(true)
    try {
      const payload = {
        title: formData.title,
        category: formData.category,
        issuer: formData.issuer,
        award_date: formData.award_date,
        description: formData.description,
        display_order: Number(formData.display_order) || 0,
        published: formData.published !== false
      }

      if (formData.id) {
        const { data, error } = await supabase
          .from('achievements')
          .update(payload)
          .eq('id', formData.id)
          .select()
          .single()

        if (error) throw error
        setItems((prev) => prev.map((i) => (i.id === formData.id ? data : i)))
        setFeedback({ type: 'success', message: `Achievement updated successfully!` })
      } else {
        const { data, error } = await supabase
          .from('achievements')
          .insert(payload)
          .select()
          .single()

        if (error) throw error
        setItems((prev) => [...prev, data])
        setFeedback({ type: 'success', message: `Achievement added successfully!` })
      }
      setModalOpen(false)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Could not save achievement: ${msg}` })
    } finally {
      setSaving(false)
    }
  }

  const handleDelete = async () => {
    if (!deletingItem) return
    setDeleting(true)
    try {
      const { error } = await supabase.from('achievements').delete().eq('id', deletingItem.id)
      if (error) throw error

      setItems((prev) => prev.filter((i) => i.id !== deletingItem.id))
      setFeedback({ type: 'success', message: `Achievement deleted successfully.` })
      setDeletingItem(null)
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Error'
      setFeedback({ type: 'error', message: `Delete failed: ${msg}` })
    } finally {
      setDeleting(false)
    }
  }

  return (
    <div className="space-y-6 max-w-6xl pb-12">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 pb-4 border-b border-slate-800">
        <div>
          <h1 className="text-2xl font-bold text-white flex items-center gap-2">
            <Trophy className="w-6 h-6 text-amber-400" />
            Achievements Manager
          </h1>
          <p className="text-xs text-slate-400 mt-1">
            Manage honors, excellence awards, hackathon recognitions, and community commendations.
          </p>
        </div>

        <div className="flex items-center gap-2">
          <button
            type="button"
            onClick={fetchData}
            className="px-3 py-2 rounded-xl bg-slate-900 border border-slate-800 hover:bg-slate-800 text-slate-300 text-xs font-semibold flex items-center gap-1.5 transition-colors"
          >
            <RotateCcw className="w-3.5 h-3.5" /> Reload
          </button>
          <button
            type="button"
            onClick={handleOpenAdd}
            className="px-4 py-2 rounded-xl bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold flex items-center gap-1.5 shadow-lg shadow-blue-600/20 transition-all"
          >
            <Plus className="w-4 h-4" /> Add Achievement
          </button>
        </div>
      </div>

      {feedback && (
        <FeedbackAlert
          type={feedback.type}
          message={feedback.message}
          onClose={() => setFeedback(null)}
        />
      )}

      <div className="space-y-4">
        {loading ? (
          <FeedbackAlert type="loading" message="Loading achievements..." />
        ) : items.length === 0 ? (
          <div className="p-8 rounded-2xl bg-slate-900 border border-slate-800 text-center space-y-3">
            <Trophy className="w-10 h-10 text-slate-600 mx-auto" />
            <p className="text-sm font-semibold text-white">No achievements recorded</p>
            <p className="text-xs text-slate-400">Click &ldquo;Add Achievement&rdquo; to insert awards and honors.</p>
          </div>
        ) : (
          <div className="grid grid-cols-1 gap-4">
            {items.map((item) => (
              <div
                key={item.id}
                className={`p-5 rounded-2xl bg-slate-900 border transition-all ${
                  item.published
                    ? 'border-slate-800 hover:border-slate-700'
                    : 'border-slate-800/50 opacity-70 bg-slate-950/40'
                } flex flex-col md:flex-row md:items-start justify-between gap-4`}
              >
                <div className="space-y-2 flex-1">
                  <div className="flex items-center gap-2">
                    <span className="text-[11px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                      Order: {item.display_order}
                    </span>
                    <span className="text-[11px] font-semibold px-2 py-0.5 rounded-full bg-amber-950 text-amber-300 border border-amber-800">
                      {item.category}
                    </span>
                  </div>

                  <div>
                    <h3 className="text-base font-bold text-white">{item.title}</h3>
                    <p className="text-xs font-semibold text-blue-400">{item.issuer}</p>
                    <p className="text-[11px] text-slate-400 flex items-center gap-1.5 mt-0.5">
                      <Calendar className="w-3.5 h-3.5 text-slate-500" /> Date: {item.award_date}
                    </p>
                  </div>

                  <p className="text-xs text-slate-300 leading-relaxed">{item.description}</p>
                </div>

                <div className="flex items-center gap-2 shrink-0 pt-2 md:pt-0 border-t md:border-t-0 border-slate-800">
                  <button
                    type="button"
                    onClick={async () => {
                      const newStatus = !item.published
                      await supabase.from('achievements').update({ published: newStatus }).eq('id', item.id)
                      setItems((prev) => prev.map((i) => (i.id === item.id ? { ...i, published: newStatus } : i)))
                    }}
                    className={`px-3 py-1.5 rounded-xl text-xs font-semibold flex items-center gap-1.5 transition-colors border ${
                      item.published
                        ? 'bg-emerald-950/60 text-emerald-300 border-emerald-800'
                        : 'bg-slate-950 text-slate-500 border-slate-800'
                    }`}
                  >
                    {item.published ? <Eye className="w-3.5 h-3.5" /> : <EyeOff className="w-3.5 h-3.5" />}
                    {item.published ? 'Published' : 'Draft'}
                  </button>

                  <button
                    type="button"
                    onClick={() => handleOpenEdit(item)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-blue-950 text-blue-400 hover:text-blue-300 border border-slate-800 transition-colors"
                    title="Edit"
                  >
                    <Edit2 className="w-3.5 h-3.5" />
                  </button>

                  <button
                    type="button"
                    onClick={() => setDeletingItem(item)}
                    className="p-2 rounded-xl bg-slate-950 hover:bg-rose-950 text-rose-400 hover:text-rose-300 border border-slate-800 transition-colors"
                    title="Delete"
                  >
                    <Trash2 className="w-3.5 h-3.5" />
                  </button>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>

      <RecordModal
        isOpen={modalOpen}
        title={formData.id ? 'Edit Achievement' : 'Add Achievement'}
        subtitle="Manage award title, issuing body, category, and recognition notes"
        maxWidth="max-w-xl"
        onClose={() => setModalOpen(false)}
      >
        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-slate-300 mb-1.5">
              Award / Recognition Title <span className="text-rose-400">*</span>
            </label>
            <input
              type="text"
              required
              value={formData.title || ''}
              onChange={(e) => setFormData({ ...formData, title: e.target.value })}
              placeholder="e.g. Outstanding Data Analysis Instructor of the Year"
              className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
            />
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                Category <span className="text-rose-400">*</span>
              </label>
              <input
                type="text"

                onChange={(e) => setFormData({ ...formData, category: e.target.value })}
                placeholder="Teaching Excellence, Hackathon Winner..."
                className="w-full px-3.5 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:border-blue-500 focus:outline-none"
              />

            <div>

                Issuing Organization / Body <span className="text-rose-400">*</span>
